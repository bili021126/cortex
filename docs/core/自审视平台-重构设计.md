# Cortex 自审视平台——重构设计

**定位**：将 `cortex-self-examination.ts`（1145 行单体）拆分为可配置、可扩展的五模块架构。支持自定义输入/输出、模型/记忆/推理参数独立控制、并行实验。

---

## 一、模块拆分

```
packages/self-examination/src/
├── config.ts       ← 实验配置接口 + 默认值 + 验证
├── platform.ts     ← 组件初始化 + Agent 注册 + 工具注入
├── orchestrator.ts ← Phase 1-5 编排（甘雨规划→执行→交叉验证→共识→归档）
├── reporter.ts     ← 摘要生成 + 报告输出 + 基线对比
└── run.ts          ← CLI 入口（读配置→初始化→编排→报告）
```

### 1.1 config.ts——实验配置接口

```typescript
interface ExamConfig {
  /** 实验 ID——用于报告识别和基线对比 */
  id: string;
  /** 实验名称 */
  name: string;
  /** 任务描述——传给甘雨做规划的 prompt */
  task: string;
  /** 记忆目录——独立的记忆库路径，默认 ".cortex/self-exam" */
  memoryDir: string;
  /** 输出目录——报告和遥测输出路径，默认 "self-exam-output/<id>" */
  outputDir: string;
  /** Agent 模型覆写——null 表示使用 cortex-agents.json 默认配置 */
  agentOverrides: AgentOverrides | null;
  /** 是否并行执行——默认 true */
  parallel: boolean;
  /** 种子记忆——复制到记忆目录的初始记忆路径列表 */
  seedMemories: string[];
  /** 审计范围——全局还是只审特定包/目录 */
  scope: "full" | { packages: string[] };
  /** 收集指标 */
  metrics: ExamMetric[];
}

interface AgentOverrides {
  model: string;        // 强制所有 Agent 使用此模型
  reasoning: "none" | "low" | "high" | "max";
}

type ExamMetric = "exitCode" | "api429Count" | "apiErrorRate" | "reportFalsePositiveRate" | "eventEmitCount" | "pipelineLatency";
```

### 1.3 orchestrator.ts——Phase 1-5 编排

```typescript
async function orchestrate(config: ExamConfig, platform: Platform): Promise<ExamRun> {
  const start = Date.now();
  const events: ExamEvent[] = [];
  
  // Phase 1: 甘雨规划（基于 task 字段生成审查计划）
  const plan = await phasePlan(config, platform);
  
  // Phase 2: 执行 + 事件追踪（AuditAgent 执行审计，收集 PipelineObserver 事件）
  const auditResults = await phaseExecute(plan, platform, events);
  
  // Phase 3: 交叉验证（对比 auditResults 与实际代码状态）
  const crossCheck = await phaseCrossCheck(auditResults, config.scope);
  
  // Phase 4: 共识圆桌（仅在 auditResults 含 needsMultiPerspective 时触发）
  const consensus = plan.needsMultiPerspective ? await phaseConsensus(auditResults, platform) : null;
  
  // Phase 5: 归档（报告写入、遥测持久化）
  const archive = await phaseArchive(config, { plan, auditResults, crossCheck, consensus });
  
  return {
    config,
    startTime: start,
    endTime: Date.now(),
    events,
    plan,
    auditResults,
    crossCheck,
    consensus,
    archive,
    exitCode: 0,
  };
}
```

### 1.5 run.ts——CLI 入口

```typescript
// pnpm cortex examine --config experiments/flash-vs-think.json
// pnpm cortex verify              ← 快速门禁模式，用上次配置 + Flash 最快

async function main() {
  const configPath = process.argv[3] ?? resolveConfig("verify");
  const config = await loadConfig(configPath);
  
  const platform = initPlatform(config, process.cwd());
  const run = await orchestrate(config, platform);
  const report = generateReport(run);
  const comparison = compareToBaseline(report, config.baselinePath);
  
  printVerdict(report, comparison);
}
```
